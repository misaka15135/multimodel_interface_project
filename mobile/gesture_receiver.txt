BroadcastReceiver integration sample for GestureAccessibilityService

// In GestureAccessibilityService.kt add the receiver and register it when service connects

private val gestureReceiver = object : android.content.BroadcastReceiver() {
    override fun onReceive(context: android.content.Context?, intent: android.content.Intent?) {
        val g = intent?.getStringExtra("gesture")
        when (g) {
            "LEFT" -> performSwipeLeft()
            "RIGHT" -> performSwipeRight()
        }
    }
}

override fun onServiceConnected() {
    super.onServiceConnected()
    val filter = android.content.IntentFilter("com.gesturecontrol.GESTURE_DETECTED")
    registerReceiver(gestureReceiver, filter)
}

override fun onDestroy() {
    try { unregisterReceiver(gestureReceiver) } catch (e: Exception) {}
    super.onDestroy()
}

// Ensure GestureAccessibilityService methods performSwipeLeft/Right are accessible and use dispatchGesture as in skeleton.

Notes:
- Use LocalBroadcastManager if you want to restrict broadcasts to the app process (recommended).
- Alternatively, bind the camera service to the accessibility service for lower-latency, tighter integration.
