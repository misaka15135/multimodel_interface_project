import androidx.camera.core.CameraSelector
import androidx.camera.core.ImageAnalysis
import androidx.camera.core.ImageProxy
import androidx.camera.lifecycle.ProcessCameraProvider
import androidx.core.content.ContextCompat
import android.graphics.Bitmap
import android.graphics.ImageFormat
import android.media.Image
import android.content.Intent

private fun startCameraAnalysis() {
    val cameraProviderFuture = ProcessCameraProvider.getInstance(this)
    cameraProviderFuture.addListener(Runnable {
        val cameraProvider = cameraProviderFuture.get()
        val preview = null // not needed for headless analysis

        val analysisUseCase = ImageAnalysis.Builder()
            .setBackpressureStrategy(ImageAnalysis.STRATEGY_KEEP_ONLY_LATEST)
            .build()

        analysisUseCase.setAnalyzer(ContextCompat.getMainExecutor(this)) { image: ImageProxy ->
            val bitmap = imageToBitmap(image)
            // run detection (MediaPipe / TFLite) on bitmap
            val gesture = detectGesture(bitmap) // returns "LEFT"/"RIGHT"/null
            if (gesture != null) {
                val b = Intent("com.gesturecontrol.GESTURE_DETECTED")
                b.putExtra("gesture", gesture)
                sendBroadcast(b)
            }
            image.close()
        }

        try {
            cameraProvider.unbindAll()
            cameraProvider.bindToLifecycle(
                /* lifecycleOwner= */ this as androidx.lifecycle.LifecycleOwner,
                CameraSelector.DEFAULT_FRONT_CAMERA,
                analysisUseCase
            )
        } catch (exc: Exception) {
            // handle
        }
    }, ContextCompat.getMainExecutor(this))
}
